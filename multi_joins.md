## ASSIGNMENT ON MULTIPLE JOIN CONDITIONS:

1. WAQTD EMPLOYEE NAME AND HIS MANAGER'S NAME AND MANAGER'S MANAGER NAME IF EMPLOYEE IS WORKING AS CLERK.
```SQL
-- USING ANSI SYNTAX
SELECT E1.ENAME "EMP NAME", 
E2.ENAME "EMP'S MGR NAME", 
E3.ENAME "MGR'S MGR NAME"
FROM EMP E1 JOIN EMP E2 ON E1.MGR = E2.EMPNO
JOIN EMP E3 ON E2.MGR = E3.EMPNO;

-- USING ORACLE SYNTAX
SELECT E1.ENAME "EMP NAME", 
E2.ENAME "EMP'S MGR NAME", 
E3.ENAME "MGR'S MGR NAME"
FROM EMP E1, EMP E2, EMP E3 
WHERE E1.MGR = E2.EMPNO
AND E2.MGR = E3.EMPNO;
```

2. WAQTD EMPLOYEE NAME & HIS DNAME AND MANAGER'S DESIGNATION AND HIS LOCATION IF MANAGER WORKS IN DEPT 10 OR 20.
```SQL
-- USING ANSI SYNTAX
SELECT E1.ENAME "EMP NAME",
D1.DNAME "EMP'S DEPT NAME",
E2.JOB "EMP'S MGR JOB",
D2.LOC "MGR'S LOCATION"
FROM EMP E1 JOIN EMP E2 ON E1.MGR = E2.EMPNO
INNER JOIN DEPT D1 ON E1.DEPTNO = D1.DEPTNO
INNER JOIN DEPT D2 ON E2.DEPTNO = D2.DEPTNO
WHERE D2.DEPTNO IN (10, 20);

-- USING ORACLE SYNTAX
SELECT E1.ENAME "EMP NAME",
D1.DNAME "EMP'S DEPT NAME",
E2.JOB "EMP'S MGR JOB",
D2.LOC "MGR'S LOCATION"
FROM EMP E1, EMP E2, DEPT D1, DEPT D2
WHERE E1.MGR = E2.EMPNO
AND E1.DEPTNO = D1.DEPTNO
AND E2.DEPTNO = D2.DEPTNO
AND D2.DEPTNO IN (10, 20);
```

5. WAQTD EMP NAME & HIS DEPT NAME AND MANAGERS NAME & HIS DNAME AND MANAGERS NAME & HIS DNAME WHERE MANAGER'S MANAGER EARNS MORE THAN MANAGER SALARY.
```SQL
-- USING ANSI SYNTAX
SELECT E1.ENAME "EMP NAME",
D1.DNAME "EMP DEPT NAME",
E2.ENAME "MGR NAME",
D2.DNAME "MGR DEPT NAME",
E3.ENAME "MGR'S MGR NAME",
D3.DNAME "MGR'S MGR DEPT NAME"
FROM EMP E1 JOIN EMP E2 ON E1.MGR = E2.EMPNO
JOIN EMP E3 ON E2.MGR = E3.EMPNO
INNER JOIN DEPT D1 ON E1.DEPTNO = D1.DEPTNO
INNER JOIN DEPT D2 ON E2.DEPTNO = D2.DEPTNO
INNER JOIN DEPT D3 ON E3.DEPTNO = D3.DEPTNO
WHERE E3.SAL > E2.SAL;

-- USING ORACLE SYNTAX
SELECT E1.ENAME "EMP NAME",
D1.DNAME "EMP DEPT NAME",
E2.ENAME "MGR NAME",
D2.DNAME "MGR DEPT NAME",
E3.ENAME "MGR'S MGR NAME",
D3.DNAME "MGR'S MGR DEPT NAME"
FROM EMP E1, EMP E2, EMP E3, DEPT D1, DEPT D2, DEPT D3  
WHERE E1.MGR = E2.EMPNO
AND E2.MGR = E3.EMPNO
AND E1.DEPTNO = D1.DEPTNO
AND E2.DEPTNO = D2.DEPTNO
AND E3.DEPTNO = D3.DEPTNO
AND E3.SAL > E2.SAL;
```
